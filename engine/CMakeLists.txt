cmake_minimum_required(VERSION 3.16)
project(hauzer C)
set(CMAKE_C_STANDARD 11)

# Shared engine core (platform-independent; the Vita and Android builds reuse these files)
set(CORE src/util.c src/cel.c src/room.c src/human.c src/player.c src/render.c src/script.c src/text.c src/audio.c src/game.c src/ui.c)

find_package(SDL2 REQUIRED)
add_executable(hauzer ${CORE} src/main_pc.c)
if(TARGET SDL2::SDL2)
  target_link_libraries(hauzer PRIVATE SDL2::SDL2main SDL2::SDL2)
else()
  target_include_directories(hauzer PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(hauzer PRIVATE ${SDL2_LIBRARIES})
endif()
if(MSVC)
  target_compile_definitions(hauzer PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(hauzer PRIVATE /W3)
  set_target_properties(hauzer PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
else()
  target_compile_options(hauzer PRIVATE -Wall -Wextra -Wno-unused-parameter)
  target_link_libraries(hauzer PRIVATE m)
endif()
if(WIN32)
  target_link_libraries(hauzer PRIVATE opengl32)
elseif(APPLE)
  find_package(OpenGL REQUIRED)
  target_link_libraries(hauzer PRIVATE OpenGL::GL)
else()
  find_library(GL_LIB GL)
  if(GL_LIB)
    target_link_libraries(hauzer PRIVATE ${GL_LIB})
  endif()
endif()
